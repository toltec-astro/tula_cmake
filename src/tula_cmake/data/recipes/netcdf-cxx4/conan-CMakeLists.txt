cmake_minimum_required(VERSION 3.23)
project(netcdf-cxx4 LANGUAGES CXX)

find_package(netCDF CONFIG REQUIRED)

file(GLOB netcdf_cxx4_sources CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/cxx4/nc*.cpp")
file(GLOB netcdf_cxx4_headers CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/cxx4/nc*.h")
list(APPEND netcdf_cxx4_headers
    "${CMAKE_CURRENT_SOURCE_DIR}/cxx4/netcdf")

add_library(netcdf-cxx4 STATIC ${netcdf_cxx4_sources})
target_include_directories(netcdf-cxx4 PUBLIC
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/cxx4>"
    "$<INSTALL_INTERFACE:include>")
target_link_libraries(netcdf-cxx4 PUBLIC netCDF::netcdf)

install(TARGETS netcdf-cxx4
    ARCHIVE DESTINATION lib
    LIBRARY DESTINATION lib
    RUNTIME DESTINATION bin)
install(FILES ${netcdf_cxx4_headers} DESTINATION include)
